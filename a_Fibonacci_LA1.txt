public class LA1 {
    static int rec(int n){
        if(n<=1){
            return n;
        }
        
        return rec(n-1)+rec(n-2);
        
    }


    static int iter(int n){
        int a = 0;
        int b = 1;
        int c = 0;


        if(n==0) return 0;

        for(int i=2;i<=n;i++){
             c = a + b;
            a = b;
            b = c;
        }

        return (n==1) ? 1 : b;
    }


    public static void main(String[] args) {

        
        long start = System.nanoTime();
        int rr = rec(3);
        long end = System.nanoTime();
        long rt = end - start;

        start = System.nanoTime();
        int ir = iter(3);
        end = System.nanoTime();
        long it = end - start;


        System.out.println(rr+" "+rt);
        System.out.println(ir+" "+it);


    }
}
