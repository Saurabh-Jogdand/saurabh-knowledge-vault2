# Gradient Descent to find Local Minimum

# Function: y = (x + 3)^2

x = 2
learning_rate = 0.1
iterations = 25

for i in range(iterations):

    # Calculate gradient
    gradient = 2 * (x + 3)

    # Update x
    x = x - learning_rate * gradient

    # Calculate y
    y = (x + 3) ** 2

    print("Iteration:", i + 1, "x =", x, "y =", y)

print("\nLocal Minimum:")
print("x =", x)
print("y =", y)
