n = int(input("Number of items: "))
val = list(map(int, input("Enter values: ").split()))
wt = list(map(int, input("Enter weights: ").split()))
W = int(input("Knapsack capacity: "))
# dp[i][w] = best value using first i items with capacity w
dp = [[0] * (W + 1) for i in range(n + 1)]
for i in range(1, n + 1):
    for w in range(W + 1):
        if wt[i - 1] <= w:
            take = val[i - 1] + dp[i - 1][w - wt[i - 1]]
            skip = dp[i - 1][w]
            dp[i][w] = max(take, skip)
        else:
            dp[i][w] = dp[i - 1][w]
print("Maximum value:", dp[n][W])

def solve(row):

        return True
    for c in range(n):
        if is_safe(row, c):
            board[row][c] = 1          # place queen
